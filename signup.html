// signup.html のサインアップ処理部分 async function handleSignUp(e) { e.preventDefault(); const email = document.getElementById('email').value; const password = document.getElementById('password').value; const submitBtn = document.getElementById('submitBtn'); submitBtn.disabled = true; submitBtn.innerText = '登録中...'; const { data, error } = await supabaseClient.auth.signUp({ email: email, password: password, }); if (error) { alert('登録エラー: ' + error.message); submitBtn.disabled = false; submitBtn.innerText = 'アカウント作成'; } else { alert('アカウント作成が完了しました！ログイン画面に移動します。'); window.location.href = 'login.html'; } }
